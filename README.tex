\section{System Architecture}

\subsection{Core Components}
\begin{enumerate}
\item \textbf{MultiSigWallet.sol} - Base multi-signature wallet implementation
\item \textbf{NotaryMediatedEscrow.sol} - Main escrow contract with milestone-based releases
\item \textbf{EscrowFactory.sol} - Factory for modular escrow deployment
\item \textbf{Frontend Interface} - React-based 4-phase workflow UI
\item \textbf{Document Hashing System} - IPFS + on-chain verification
\end{enumerate}

\subsection{Key Features}
\begin{itemize}
\item \textbf{Multi-Sig Security}: Requires multiple confirmations for fund releases
\item \textbf{Notary Oracle}: Legal Notary verifies legal milestones before fund releases
\item \textbf{Document Hashing}: All legal documents hashed on-chain for verification
\item \textbf{4-Phase Workflow}: Discovery → Safety Gate → Execution → Final Settlement
\item \textbf{Escrow Factory}: Modular deployment for any land purchase
\end{itemize}

\section{4-Phase Workflow}

\subsection{Phase 1: Discovery \& Lock-up}
\begin{itemize}
\item \textbf{User Action}: Client deposits total budget (e.g., \$5,000) into Treasury
\item \textbf{System Action}: Funds are locked in multi-sig
\item \textbf{UI Component}: Curated list of ``Digital Proofs'' (Geo-tagged photos/videos)
\end{itemize}

\subsection{Phase 2: Safety Gate (\$200 Due Diligence)}
\begin{itemize}
\item \textbf{Condition}: First milestone of \$200 unlocked for plot selection
\item \textbf{Backend Logic}: Goes to Legal Notary's wallet for Registry Search and site visit
\item \textbf{UI Component}: ``Document Vault'' for viewing Registry Search and Site Visit Report
\end{itemize}

\subsection{Phase 3: Execution (Commitment Milestone)}
\begin{itemize}
\item \textbf{Condition}: Both parties sign draft agreement on-platform
\item \textbf{On-Platform Signing}: Client or seller uploads draft, both parties sign with real-time notifications
\item \textbf{Notary Role}: Legal Notary verifies signed documents and clicks ``Proceed to Next Step''
\item \textbf{Smart Contract Action}: Release percentage-based deposit to seller after notarization
\item \textbf{UI Component}: Document signing system with party status tracking
\end{itemize}

\subsection{Phase 4: Final Settlement \& Notary Release}
\begin{itemize}
\item \textbf{Condition}: Legal Notary confirms Original Title and LCB Consents
\item \textbf{Smart Contract Action}: Release balance to seller + facilitation fee.
\item \textbf{UI Component}: ``Finalization Status'' showing transfer progress
\end{itemize}

\section{Smart Contract Details}

\subsection{NotaryMediatedEscrow.sol - Key Functions}
\begin{verbatim}
// Create new escrow
function createEscrow(address _seller, address _notary, uint _totalAmount) returns (uint)

// Fund escrow with total amount
function fundEscrow(uint _escrowId) payable

// Upload draft sale agreement (client or seller)
function uploadDraftAgreement(uint _escrowId, bytes32 _hash)

// Sign document on platform (both parties)
function signDocument(uint _escrowId, DocumentType _docType)

// Notarize signed document and proceed (notary only)
function notarizeSignedDocument(uint _escrowId, DocumentType _docType)

// Upload other documents (notary only)
function uploadDocument(uint _escrowId, DocumentType _docType, bytes32 _hash)

// Finalize transfer (notary only)
function finalizeTransfer(uint _escrowId)
\end{verbatim}

\subsection{Milestone Structure}
\begin{verbatim}
enum Milestone { 
    DueDiligence,    // $200 for registry search and site visit
    Deposit,         // Percentage-based deposit to seller
    FinalSettlement  // Balance + facilitation fee
}
\end{verbatim}

\begin{verbatim}

    OriginalTitle, 
    LCBConsent

enum SigningStatus { 
    NotStarted, 
    Draft, 
    ClientSigned, 
    SellerSigned, 
    BothSigned, 
    NotaryVerified 
}
\end{verbatim}

\section{Financial Flow}

\subsection{Fee Structure}
\begin{verbatim}
Total Purchase Amount: $5,000
├── Due Diligence Fee: $200 (4%)
├── Deposit (10%): $500
├── Final Balance: $4,200
└── Facilitation Fee: $100 (2% of total)
\end{verbatim}

\subsection{Release Schedule}
\begin{enumerate}
\item \$200 → Legal Notary (Due Diligence)
\item \$500 → Seller (Deposit after Sale Agreement)
\item \$4,200 → Seller (Final Balance after Title Transfer)
\item \$100 → Facilitation Fee
\end{enumerate}
